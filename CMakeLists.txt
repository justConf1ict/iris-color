cmake_minimum_required(VERSION 3.20)
project(IrisColor VERSION 0.7.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(IRIS_ENABLE_ASAN "Instrument native MSVC builds with AddressSanitizer" OFF)
if(IRIS_ENABLE_ASAN)
    if(NOT MSVC)
        message(FATAL_ERROR "IRIS_ENABLE_ASAN currently supports the MSVC Windows build")
    endif()
    add_compile_options(/fsanitize=address /Zi)
    add_link_options(/INCREMENTAL:NO)
endif()
set(IRIS_VST3_SDK "${CMAKE_CURRENT_SOURCE_DIR}/third_party/vst3sdk" CACHE PATH "Pinned VST3 SDK source directory")
if(NOT EXISTS "${IRIS_VST3_SDK}/public.sdk/source/vst/vstaudioeffect.cpp")
    message(FATAL_ERROR "VST3 SDK missing. Run python tools/native_build_bootstrap.py first.")
endif()
if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Iris Color is a 64-bit plugin. Configure Visual Studio with -A x64.")
endif()

file(STRINGS "plugin/sdk_sources.txt" IRIS_SDK_RELATIVE)
set(IRIS_SDK_SOURCES)
foreach(source IN LISTS IRIS_SDK_RELATIVE)
    list(APPEND IRIS_SDK_SOURCES "${IRIS_VST3_SDK}/${source}")
endforeach()
add_library(iris_vst3_sdk STATIC ${IRIS_SDK_SOURCES})
target_include_directories(iris_vst3_sdk PUBLIC "${IRIS_VST3_SDK}")
target_compile_definitions(iris_vst3_sdk PUBLIC "$<$<CONFIG:Debug>:DEVELOPMENT=1>" "$<$<NOT:$<CONFIG:Debug>>:RELEASE=1>")
if(MSVC)
    target_compile_options(iris_vst3_sdk PUBLIC /utf-8 /EHsc)
endif()

add_library(iris_dsp STATIC src/IrisDSP.cpp)
target_include_directories(iris_dsp PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(MSVC)
    target_compile_options(iris_dsp PUBLIC /utf-8 /EHsc)
endif()
add_library(IrisColor MODULE plugin/IrisParameters.cpp plugin/IrisPlugin.cpp plugin/IrisEditorWin.cpp plugin/IrisEntry.cpp)
target_include_directories(IrisColor PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(IrisColor PRIVATE iris_dsp iris_vst3_sdk)
set_target_properties(IrisColor PROPERTIES PREFIX "" OUTPUT_NAME "Iris Color")
if(WIN32)
    enable_language(RC)
    target_sources(IrisColor PRIVATE "${IRIS_VST3_SDK}/public.sdk/source/main/dllmain.cpp" plugin/IrisColor.def plugin/IrisVersion.rc)
    target_compile_definitions(iris_vst3_sdk PUBLIC UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
    target_link_libraries(IrisColor PRIVATE ole32 uuid user32 gdi32 gdiplus shell32 winmm)
    set_target_properties(IrisColor PROPERTIES SUFFIX ".vst3")
    set(IRIS_ARCH_DIR "x86_64-win")
else()
    target_sources(IrisColor PRIVATE "${IRIS_VST3_SDK}/public.sdk/source/main/linuxmain.cpp")
    target_link_libraries(IrisColor PRIVATE pthread dl)
    set(IRIS_ARCH_DIR "x86_64-linux")
endif()
set(IRIS_BUNDLE "${CMAKE_BINARY_DIR}/$<CONFIG>/Iris Color.vst3")
set_target_properties(IrisColor PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${IRIS_BUNDLE}/Contents/${IRIS_ARCH_DIR}"
    LIBRARY_OUTPUT_DIRECTORY "${IRIS_BUNDLE}/Contents/${IRIS_ARCH_DIR}")
add_custom_command(TARGET IrisColor POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${IRIS_BUNDLE}/Contents/Resources"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${IRIS_VST3_SDK}/LICENSE.txt" "${IRIS_BUNDLE}/Contents/Resources/VST3-SDK-LICENSE.txt"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts" "${IRIS_BUNDLE}/Contents/Resources/fonts")
file(GLOB IRIS_THIRD_PARTY_NOTICES "${CMAKE_CURRENT_SOURCE_DIR}/third_party/*LICENSE.txt")
foreach(notice IN LISTS IRIS_THIRD_PARTY_NOTICES)
    add_custom_command(TARGET IrisColor POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${notice}" "${IRIS_BUNDLE}/Contents/Resources/")
endforeach()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
    add_custom_command(TARGET IrisColor POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" "${IRIS_BUNDLE}/Contents/Resources/IRIS-COLOR-LICENSE.txt")
elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE.txt")
    add_custom_command(TARGET IrisColor POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE.txt" "${IRIS_BUNDLE}/Contents/Resources/IRIS-COLOR-LICENSE.txt")
endif()

add_executable(iris-render tools/render.cpp)
target_link_libraries(iris-render PRIVATE iris_dsp)

option(IRIS_BUILD_REGRESSION_TESTS "Build native regression executables" OFF)
if(IRIS_BUILD_REGRESSION_TESTS)
    enable_testing()
    add_library(iris_parameters STATIC plugin/IrisParameters.cpp)
    target_include_directories(iris_parameters PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(iris_parameters PUBLIC iris_vst3_sdk)
    set(IRIS_HOST_SUPPORT
        "${IRIS_VST3_SDK}/public.sdk/source/vst/hosting/hostclasses.cpp"
        "${IRIS_VST3_SDK}/public.sdk/source/vst/hosting/pluginterfacesupport.cpp"
        "${IRIS_VST3_SDK}/public.sdk/source/vst/utility/stringconvert.cpp"
        "${IRIS_VST3_SDK}/public.sdk/source/common/commonstringconvert.cpp")
    foreach(name gui_lifecycle_test gui_controls_test stability_test host_contract_test host_test transport04_test)
        add_executable(${name} "plugin/tests/${name}.cpp" ${IRIS_HOST_SUPPORT})
        target_link_libraries(${name} PRIVATE iris_parameters ole32 uuid user32 gdi32 gdiplus shell32 winmm)
    endforeach()
    foreach(name state_compat_test state_compat04_test state_compat05_test state_compat06_test state_compat07_test)
        add_executable(${name} "plugin/tests/${name}.cpp")
        target_link_libraries(${name} PRIVATE iris_parameters)
        add_test(NAME ${name} COMMAND ${name})
    endforeach()
    # dsp_color04_neutral needs a separately supplied v0.3 reference header.
    foreach(name dsp_unit dsp_color03_bands dsp_color03_level dsp_color03_state dsp_color04_contracts dsp_color04_features dsp_color05_dive_guard dsp_color05_mode dsp_color07_body dsp_revision_visual dsp04_review_extremes visual_mailbox)
        add_executable(${name} "tests/${name}.cpp")
        target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
        target_link_libraries(${name} PRIVATE iris_dsp)
        add_test(NAME ${name} COMMAND ${name})
    endforeach()
endif()
